\begin{frame}{Why Stream Processing?}
\begin{columns}

  \textbf{The Problem with Batch}
  \begin{itemize}
    \item \textbf{High latency} — jobs run hourly/daily, insights lag by hours
    \item \textbf{Stale decisions} — by the time data is ready, the moment to act has passed
    \item \textbf{Resource spikes} — large periodic loads strain infra, then sit idle
  \end{itemize}

\section{Background \& Design Choices}
%=============================================================================

\begin{frame}{Core Concepts}
\begin{columns}
  \column{0.48\textwidth}
  \textbf{Kafka: Publish/Subscribe}
  \begin{itemize}\small
    \item Producers write to \textbf{topics}; consumers subscribe independently
    \item Durable, replicated, append-only log — replay any offset
  \end{itemize}
  \vspace{0.2cm}
  \textbf{Structured Streaming: Micro-Batch}
  \begin{itemize}\small
    \item Treats a stream as an unbounded, incrementally-processed table
    \item Each trigger reads a bounded slice of new Kafka records and runs a Spark batch job over it
  \end{itemize}

  \column{0.48\textwidth}
  \textbf{Hudi: COW vs.\ MOR}
  \begin{itemize}\small
    \item \textbf{Copy-on-Write (COW)} — rewrites base files on every write; fast reads, simple
    \item \textbf{Merge-on-Read (MOR)} — appends to delta log files, compacted later; fast writes for high-frequency streaming appends
  \end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Why Spark + Hudi + Superset?}
  \begin{itemize}\setlength\itemsep{6pt}
    \item \textbf{Spark Structured Streaming vs.\ Flink} — one engine (Spark) covers both the streaming ingest job and the batch SQL that dbt submits, so the same 4-core cluster serves both without running two separate engines
    \item \textbf{Hudi vs.\ Snowflake} — an open table format on commodity object storage (MinIO/S3A); no managed warehouse subscription, and MOR bronze tables absorb frequent small streaming writes cheaply
    \item \textbf{Superset vs.\ Power BI} — open-source, connects directly to the same Spark Thrift Server the pipeline already runs, no extra licensing or export step
  \end{itemize}
\end{frame}

\section{Problem \& Challenges}
%=============================================================================

\begin{frame}{Dataset}
  \textbf{Eventsim}
  \begin{itemize}\small
    \item Simulates page requests for a fake music-streaming website — synthetic but realistic user behaviour
    \item Generates three event streams: \texttt{listen\_events}, \texttt{page\_view\_events}, \texttt{auth\_events}
  \end{itemize}
  \vspace{0.2cm}
  \textbf{Million Song Dataset (subset)}
  \begin{itemize}\small
    \item Eventsim draws song/artist metadata from a \textbf{10,000-song subset} of the Million Song Dataset
    \item Gives listen events realistic song, artist, and duration attributes
  \end{itemize}
\end{frame}

\begin{frame}{Technical Challenges}
  \begin{itemize}\setlength\itemsep{6pt}
    \item \textbf{Continuous streaming volume} — events must be ingested, decoded, and enriched (mojibake fixes, derived time columns) without falling behind Kafka
    \item \textbf{Small, shared compute} — the whole Spark Standalone cluster is \textbf{1 master + 2 workers, 4 cores total}, split between the streaming job and the dbt Thrift Server, with no headroom for ad-hoc SQL
    \item \textbf{Partitioning \& parallelism} — Kafka topics have only 3 partitions each; partition count, replication factor, and Spark's capped cores all bound end-to-end throughput
  \end{itemize}
\end{frame}

\section{Message Queue: Apache Kafka}
%=============================================================================

\begin{frame}{Why Apache Kafka?}
\begin{columns}
  \column{0.48\textwidth}
  \textbf{Problems Without a Message Queue}
  \begin{itemize}
    \item \textbf{Tight coupling} — producer knows every consumer
    \item \textbf{No buffering} — slow consumer $\to$ data loss
    \item \textbf{No replay} — consumed = gone forever
    \item \textbf{Fan-out cost} — $N$ consumers = $N$ direct calls
  \end{itemize}

  \column{0.48\textwidth}
  \textbf{How Kafka Solves It}
  \begin{itemize}
    \item \textbf{Decoupling} — write to topic, subscribe independently
    \item \textbf{Durable buffer} — persisted to disk, replicated
    \item \textbf{Replay} — configurable retention, re-read anytime
    \item \textbf{Scalability} — partitions scale throughput linearly
    \item \textbf{Fan-out free} — multiple groups, own offsets
  \end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Kafka Architecture}
\begin{columns}
  \column{0.52\textwidth}
  \includegraphics[width=\textwidth]{kafka/kafka_architecture.png}

  \column{0.45\textwidth}
  \begin{itemize}
    \item \textbf{Producer} — partitions by key; \texttt{acks=all} for durability
    \item \textbf{Broker} — leader/follower replicas; append-only log
    \item \textbf{Topic} — immutable records; retention-based or compacted
    \item \textbf{Partition} — ordered, offset-indexed; unit of parallelism
    \item \textbf{Consumer Group} — one partition $\to$ one member; independent offset per group
  \end{itemize}
  \vspace{0.2cm}
  \small\textit{Max parallelism = number of partitions}
\end{columns}
\end{frame}

\begin{frame}{Kafka Setup in Streamify}
\begin{columns}
  \column{0.48\textwidth}
  \textbf{2-Node KRaft Cluster}
  \begin{itemize}\small
    \item Runs on \textbf{apache/kafka:4.3.0} — no Zookeeper
    \item Each node (\texttt{broker-1}, \texttt{broker-2}) is both broker \& controller
    \item \textbf{Replication factor 2} — every partition has a replica on both brokers, no data loss if one broker restarts
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Topic Configuration}
  \begin{itemize}\small
    \item 3 topics: \texttt{listen\_events}, \texttt{page\_view\_events}, \texttt{auth\_events}
    \item \textbf{3 partitions} per topic — enables parallel consumption by Spark Structured Streaming
    \item Topics pre-created by \texttt{kafka-init} before producers start
  \end{itemize}

\section{Stream Processing: Spark Structured Streaming \& Hudi}
%=============================================================================

\begin{frame}{Spark Structured Streaming: Micro-Batch Model}
\begin{columns}
  \column{0.55\textwidth}
  \includegraphics[width=\textwidth]{engine/spark_micro_batch.png}
  \par\vspace{1pt}{\tiny Each trigger processes a bounded batch of new Kafka records as a Spark job}
  \vspace{0.15cm}

  \includegraphics[width=\textwidth]{engine/spark_architecture.png}

  \column{0.42\textwidth}
  \textbf{Dataflow Model}
  \begin{itemize}\small
    \item \textbf{Micro-batch} — the stream is an unbounded table, incrementally queried on a trigger interval
    \item Same Spark engine and SQL/DataFrame API as batch jobs
    \item Result: each batch's rows appended straight to Hudi
  \end{itemize}
  \vspace{0.25cm}
  \textbf{Fault Tolerance}
  \begin{itemize}\small
    \item Structured Streaming \textbf{checkpoint} tracks processed Kafka offsets and query state
    \item Persisted to the \texttt{spark-checkpoints} Docker volume — resumes from last committed offset on restart
    \item Offsets also committed to a \texttt{streamify-<topic>} consumer group purely so lag is visible in Grafana (Spark itself does not depend on that commit to resume)
  \end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Writing to Hudi: Bronze Layer}
\begin{columns}
  \column{0.48\textwidth}
  \textbf{MERGE\_ON\_READ Bronze Tables}
  \begin{itemize}\small
    \item Each micro-batch appends to \texttt{staging.<topic>} as a Hudi \textbf{MERGE\_ON\_READ (MOR)} table
    \item Fast streaming appends land as delta log files, periodically compacted into columnar base files
    \item Backed by MinIO (dev) or S3 (\texttt{aws} mode) via the Hadoop S3A connector — no rebuild needed to switch
    \item No separate catalog service — each table self-describes via its \texttt{.hoodie/} metadata directory, (re-)registered into Spark's catalog at job startup
  \end{itemize}

  \column{0.48\textwidth}
  \includegraphics[width=\textwidth]{engine/spark_micro_batch_latency.png}
\end{columns}
\end{frame}

\begin{frame}{Spark Setup in Streamify}
\begin{columns}
  \column{0.48\textwidth}
  \textbf{Shared Standalone Cluster}
  \begin{itemize}\small
    \item \textbf{1 master + 2 workers}, 2 cores / 2 GB each — 4 cores total
    \item \textbf{spark-streaming} job (Kafka $\to$ Hudi bronze), capped at 2 cores
    \item \textbf{spark-thrift} Thrift Server (dbt SQL submission), capped at 2 cores
    \item Full cluster used by the two long-running applications — no headroom for ad-hoc Spark SQL at this size
  \end{itemize}
  \vspace{0.2cm}
  \textbf{Streaming Pipeline}
  \begin{itemize}\small
    \item Source: Kafka connector (3 topics)
    \item Transform: fix latin1-mojibake on text fields, derive time columns
    \item Sink: Hudi bronze tables, MERGE\_ON\_READ
  \end{itemize}

  \column{0.48\textwidth}
  \includegraphics[width=\textwidth]{engine/dataflow.png}
\end{columns}
\end{frame}

\section{Conclusion}
%=============================================================================

\begin{frame}{Conclusion}
\textbf{Results Achieved}
\begin{enumerate}\setlength\itemsep{4pt}
  \item \textbf{Kafka} decouples producers from consumers — 2-node KRaft cluster, RF2, 3 partitions per topic gives fault tolerance without a third broker
  \item \textbf{Spark Structured Streaming} ingests all 3 topics as micro-batches, enriches them, and appends to \textbf{Hudi MERGE\_ON\_READ} bronze tables, resuming from a persisted checkpoint on restart
  \item \textbf{ELT pattern} keeps concerns separate: Spark loads raw events into Hudi, dbt transforms in place into a star schema and wide table via the Spark Thrift Server
  \item \textbf{Superset} dashboards and \textbf{Prometheus/Grafana} monitoring close the loop — data is queryable and the pipeline's health is visible in real time
\end{enumerate}

\vspace{0.2cm}
\textbf{Limitations}
\begin{itemize}\small
  \item The 4-core Spark Standalone cluster is shared between streaming ingest and the dbt Thrift Server, leaving no headroom for ad-hoc SQL
  \item Gold-layer dbt models rebuild via full overwrite rather than incremental merge
  \item No recorded throughput/latency benchmarks yet (see Evaluation)
\end{itemize}

\vspace{0.2cm}
\textbf{Future Directions}
\begin{itemize}\small
  \item Scale the Spark cluster (more workers/cores) to separate streaming and Thrift-server workloads
  \item Move gold dbt models to incremental materializations
  \item Add schema evolution / validation for the incoming Eventsim payloads
\end{itemize}
\end{frame}
