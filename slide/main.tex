\section{Introduction}
%=============================================================================

\begin{frame}{Motivation \& Goals}
\begin{columns}
  \column{0.5\textwidth}
  \textbf{Why a streaming lakehouse}
  \begin{itemize}\small
    \item Eventsim emits a continuous stream of song-play, page-view, and login events onto 3 Kafka topics, not daily batches
    \item Batch ETL is always some interval behind reality -- fine for reports, too slow for "what is happening now"
    \item Doing this reliably (not just fast) means solving ordering, fault tolerance, exactly-once delivery, and read/write contention on a shared 4-core Spark cluster
  \end{itemize}

  \column{0.47\textwidth}
  \textbf{Goals}
  \begin{itemize}\small
    \item Build the full Eventsim $\to$ Kafka $\to$ Spark Structured Streaming $\to$ Hudi $\to$ dbt $\to$ Superset pipeline, end to end
    \item Apply distributed-systems concepts to a real system: 2-node Kafka replication, checkpointed exactly-once processing, consumer-lag monitoring
    \item Make the whole system observable (Prometheus/Grafana) and reproducible -- single Docker Compose command
  \end{itemize}
\end{columns}
\end{frame}

%=============================================================================
\section{Problem \& Challenges}
%=============================================================================

\begin{frame}{The Problem Before Streaming Lakehouses}
  \begin{itemize}\setlength\itemsep{2pt}\small
    \item \textbf{Batch ETL latency} -- dashboards always reflect the last scheduled load, not the current state
    \item \textbf{Lambda-architecture complexity} -- a batch layer and a separate real-time speed layer duplicate the same logic in two codebases
    \item \textbf{No serving layer on the lake} -- plain object storage has no row-level update/delete or ACID transactions; the lake stayed a read-only dump
    \item \textbf{Storage and compute coupled} -- traditional warehouses force paying for both together, so teams kept two copies of the same data
  \end{itemize}
  \vspace{0.2cm}
  \textbf{This project's concrete challenges:}
  \begin{itemize}\small
    \item \textbf{Streaming volume \& velocity} -- keeping up with Eventsim's continuous arrival rate across \texttt{listen\_events}, \texttt{page\_view\_events}, \texttt{auth\_events}
    \item \textbf{Exactly-once, fault-tolerant processing} on a 2-node KRaft cluster (replication factor 2, 3 partitions/topic), no ZooKeeper
    \item \textbf{Compute and partitioning} on a 4-core Spark Standalone cluster (1 master + 2 workers, 2 cores/2\,GB each), shared 2 cores each between the streaming job and the dbt Thrift Server
  \end{itemize}
\end{frame}

\section{Proposed Solution}
%=============================================================================

\begin{frame}{Streamify: End-to-End Architecture}
\centering
\includegraphics[width=1\textwidth]{overview.png}
\end{frame}

\begin{frame}{Ingestion Layer: Message Queues \& Apache Kafka}
\centering
\includegraphics[width=0.5\textwidth]{message_queue.png}
\vspace{0.2cm}
\begin{itemize}\small
  \item \textbf{Message queue (pub-sub)}: producers append to a durable, ordered log; consumers subscribe independently, at their own pace -- decoupled in both time and number
  \item It play a role like a buffer between the producer and consumer, smoothing out bursts of traffic and allowing consumers to catch up
  \item 2-node KRaft cluster (\texttt{broker-1}, \texttt{broker-2}) -- no ZooKeeper, each node is broker \& controller
  \item 3 topics (\texttt{listen\_events}, \texttt{page\_view\_events}, \texttt{auth\_events}), 3 partitions each, replication factor 2 -- every partition has a leader on one broker and a follower on the other
\end{itemize}
\end{frame}

\begin{frame}{Streaming vs.\ Batch Processing}
\begin{columns}
  \column{0.48\textwidth}
  \centering
  \includegraphics[width=\textwidth]{spark_batch_streaming_processing.png}
  \column{0.48\textwidth}
  \begin{itemize}\small
    \item \textbf{Streaming processing} -- Spark Structured Streaming reads all Kafka topics continuously, parses and appends straight into storage as bronze (Hudi MERGE\_ON\_READ)
    \item Checkpoint volume tracks Kafka offsets and query state -- resumes exactly where it left off on restart
    \item \textbf{Batch processing} -- Spark also runs as the pipeline's transform layer: dbt reads bronze on a fixed trigger and builds the silver/gold models
  \end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Storage Layer \& Open Table Formats}
\centering
\includegraphics[width=0.5\textwidth]{storage_open_table_format.png}
\vspace{0.2cm}
\begin{itemize}\small
  \item \textbf{Distributed storage} (HDFS, S3, GCS, Azure) $\to$ \textbf{open file format} (Parquet, ORC) $\to$ \textbf{open table format} (Hudi, Delta Lake, Iceberg) on top, for ACID upserts and time travel
  \item Plain files on distributed storage cannot do row-level update/delete or ACID transactions -- the open table format adds that, so the lake behaves like a table instead of an unmanaged pile of files
  \item COPY\_ON\_WRITE -- fast reads, expensive writes; MERGE\_ON\_READ -- cheap streaming appends, merged at read time
\end{itemize}
\end{frame}

\begin{frame}{Transformation Layer: dbt}
\centering
\includegraphics[width=0.5\textwidth]{dbt_graph.png}
\vspace{0.2cm}
\begin{itemize}\small
  \item \textbf{Silver} -- 5 dimension tables from bronze: \texttt{dim\_users}, \texttt{dim\_songs}, \texttt{dim\_artists}, \texttt{dim\_locations}, \texttt{dim\_datetime}
  \item \textbf{Gold} -- \texttt{fact\_streams} (bronze + 5 dims, star schema) and \texttt{wide\_streams} (\texttt{fact\_streams} pre-joined with all 5 dims, so Superset needs no dashboard-time joins)
  \item dbt is a SQL layer over the same Spark cluster: it generates Spark SQL and submits it via the Spark Thrift Server, materializing every model as COPY\_ON\_WRITE Hudi, rebuilt via full overwrite each run
\end{itemize}
\end{frame}

\begin{frame}{Serving: Apache Superset}
\centering
\includegraphics[width=0.8\textwidth]{dashboard.png}
\vspace{0.2cm}
\begin{itemize}\small
  \item Superset queries \texttt{wide\_streams} through the Spark Thrift Server -- one flat table, no dashboard-time joins
  \item \textbf{KPIs}: total streams, total users. \textbf{Breakdowns}: top songs \& artists (Chart Busters, Top Artists), gender distribution, user level by gender
  \item \textbf{Time series}: user activity (stream count per hour). \textbf{Geo}: listen events per US state
\end{itemize}
\end{frame}

\begin{frame}{Monitoring Layer}

\begin{columns}
  \column{0.48\textwidth}
  \centering
  \includegraphics[width=\textwidth]{grafana_kafka.png}
  \column{0.48\textwidth}
  \centering

\begin{itemize}\small
  \item \textbf{Kafka metrics} (kafka-exporter) -- consumer-group lag per partition, messages in/out per second, under-replicated partitions
  \item \textbf{Spark metrics} (Prometheus endpoint) -- streaming batch duration \& processing rate, executor CPU/memory, active jobs/stages
  \item \textbf{Container metrics} (cAdvisor) -- per-container CPU and memory usage across the cluster
  \item Grafana dashboards make these visible in real time -- e.g.\ rising consumer lag or batch duration flags the pipeline falling behind, not just whether it is running
\end{itemize}

\end{frame}

\section{Conclusion}
%=============================================================================

\begin{frame}{Conclusion}
\textbf{Results Achieved}
\begin{itemize}\setlength\itemsep{4pt}\small
  \item Understood, hands-on, how a streaming platform works end to end: why a Kafka partition needs a follower replica, why the Structured Streaming checkpoint is separate from the consumer-group offset, when MERGE\_ON\_READ is worth its read-time cost
  \item Exercised partitioning, replication, checkpointed exactly-once processing, and lag observability against real failure modes, not just in theory
  \item Confirmed why the streaming-lakehouse pattern closes the batch/lambda-architecture gaps: no duplicated speed-layer logic, a transactionally-updatable lake, storage and compute that scale independently
\end{itemize}

\vspace{0.2cm}
\textbf{Limitations}
\begin{itemize}\small
  \item Small, fixed 4-core Spark cluster, no autoscaling
  \item Synthetic Eventsim dataset, not validated against real production traffic
\end{itemize}

\vspace{0.2cm}
\textbf{Future Directions}
\begin{itemize}\small
  \item Real or real-scale streaming dataset; autoscaled Kafka/Spark cluster
  \item Alerting on consumer lag and job failures
\end{itemize}
\end{frame}
